 Relais SRD-05VDC-SL-C : IN -> D7, VCC -> 5V, GND -> GND
 Côté 12 V : +12V -> COM ; NO -> hublot VERT ; NC -> hublot ROUGE ;
             autre borne des deux hublots -> 0V

---------------------------------------------------------------------
 A) PROGRAMME DE TEST DU RELAIS SEUL (sans le capteur)
    À téléverser seul pour vérifier le câblage :
    le relais doit claquer et les hublots alterner toutes les 2 s.
---------------------------------------------------------------------

#define RELAIS        7      // broche IN du module relais
#define RELAIS_ACTIF  HIGH   // mettre LOW si le vert reste allumé au repos
#define RELAIS_REPOS  (RELAIS_ACTIF == HIGH ? LOW : HIGH)

void setup() {
  digitalWrite(RELAIS, RELAIS_REPOS);
  pinMode(RELAIS, OUTPUT);
  Serial.begin(9600);
  Serial.println(F("Test du relais"));
}

void loop() {
  digitalWrite(RELAIS, RELAIS_ACTIF);
  Serial.println(F("Relais ACTIF : hublot VERT"));
  delay(2000);

  digitalWrite(RELAIS, RELAIS_REPOS);
  Serial.println(F("Relais REPOS : hublot ROUGE"));
  delay(2000);
}
